    Set c = parent.Controls.Add(progId)

    c.Font.Size = 9
    c.ForeColor = CLR_TEXT

    c.BackStyle = fmBackStyleOpaque

End Function

Private Sub scrBright_Change()

Private Sub SyncBrightText()
    If mUpdating Then Exit Sub
    mUpdating = True
    txtBright.Text = CStr(scrBright.Value)
    mUpdating = False
End Sub

Private Sub txtBright_Change()
    Dim v As Long
    If mUpdating Then Exit Sub
    If Not IsNumeric(txtBright.Text) Then Exit Sub
    v = CLng(Val(txtBright.Text))
    If v < -100 Then v = -100
    If v > 100 Then v = 100
    mUpdating = True
    scrBright.Value = v
    mUpdating = False
End Sub

' =====================================================================
'  ASOSIY ISH
' =====================================================================
Private Sub btnRun_Click()
    Dim res As Long, mode As cdrImageType, bright As Long
    Dim pg As Page, sr As ShapeRange
    Dim okCount As Long, errCount As Long

    If ActiveDocument Is Nothing Then
        MsgBox "Ochiq hujjat yo'q.", vbExclamation, Me.Caption
        Exit Sub
    End If

    If Not IsNumeric(cboRes.Text) Then
        MsgBox "Razreshenie (DPI) raqam bo'lishi kerak.", vbExclamation, Me.Caption
        Exit Sub
    End If
    res = CLng(Val(cboRes.Text))
    If res < 30 Or res > 3000 Then
        MsgBox "Razreshenie 30 dan 3000 gacha bo'lishi kerak.", vbExclamation, Me.Caption
        Exit Sub
    End If

    If optRGB.Value Then
        mode = cdrRGBColorImage
    ElseIf optCMYK.Value Then
        mode = cdrCMYKColorImage
    Else
        mode = cdrGrayscaleImage
    End If
    bright = scrBright.Value

    If optSel.Value And ActiveSelectionRange.Count = 0 Then
        MsgBox "Hech narsa tanlanmagan.", vbExclamation, Me.Caption
        Exit Sub
    End If

    btnRun.Enabled = False
    ActiveDocument.BeginCommandGroup "Bitmap konverter"
    Optimization = True
    On Error GoTo Finish

    If optSel.Value Then
        Set sr = FilterRange(ActiveSelectionRange)
        ProcessRange sr, mode, res, bright, okCount, errCount
    ElseIf optPage.Value Then
        Set sr = CollectPage(ActivePage)
        ProcessRange sr, mode, res, bright, okCount, errCount
    Else
        For Each pg In ActiveDocument.Pages
            pg.Activate
            SetStatus "List " & pg.Index & " / " & ActiveDocument.Pages.Count & " ..."
            Set sr = CollectPage(pg)
            ProcessRange sr, mode, res, bright, okCount, errCount
        Next pg
    End If

Finish:
    If Err.Number <> 0 Then errCount = errCount + 1
    On Error GoTo 0
    Optimization = False
    ActiveDocument.EndCommandGroup
    ActiveWindow.Refresh
    Application.Refresh
    btnRun.Enabled = True

    SetStatus "Tayyor: " & okCount & " ta bitmap qilindi" & _
              IIf(errCount > 0, ", " & errCount & " ta xato.", ".")
    MsgBox okCount & " ta obyekt bitmapga aylantirildi." & _
           IIf(errCount > 0, vbCrLf & errCount & " ta obyektda xato bo'ldi.", ""), _
           vbInformation, Me.Caption
End Sub

' Listdagi barcha mos obyektlarni yig'ish (yashirin/qulflangan qatlamlar o'tkazib yuboriladi)
Private Function CollectPage(pg As Page) As ShapeRange
    Dim sr As ShapeRange, lyr As Layer, s As Shape
    Set sr = CreateShapeRange
    For Each lyr In pg.Layers
        If Not lyr.IsSpecialLayer And lyr.Visible And lyr.Editable Then
            For Each s In lyr.Shapes
                If IsEligible(s) Then sr.Add s
            Next s
        End If
    Next lyr
    Set CollectPage = sr
End Function

Private Function FilterRange(src As ShapeRange) As ShapeRange
    Dim sr As ShapeRange, s As Shape
    Set sr = CreateShapeRange
    For Each s In src
        If IsEligible(s) Then sr.Add s
    Next s
    Set FilterRange = sr
End Function

Private Function IsEligible(s As Shape) As Boolean
    If s.Type = cdrGuidelineShape Then Exit Function
    If s.Locked Then Exit Function
    If chkOnlyBmp.Value Then
        IsEligible = (s.Type = cdrBitmapShape)
    Else
        IsEligible = True
    End If
End Function

Private Sub ProcessRange(sr As ShapeRange, mode As cdrImageType, res As Long, bright As Long, _
                         okCount As Long, errCount As Long)
    Dim i As Long
    If sr.Count = 0 Then Exit Sub

    If optOne.Value And sr.Count > 1 Then
        If ConvertShape(sr.Group, mode, res, bright) Then okCount = okCount + 1 Else errCount = errCount + 1
    Else
        For i = 1 To sr.Count
            SetStatus "Ishlanmoqda: " & i & " / " & sr.Count
            If ConvertShape(sr(i), mode, res, bright) Then okCount = okCount + 1 Else errCount = errCount + 1
        Next i
    End If
End Sub

' Bitta obyektni bitmapga aylantirish.
' Yorqinlik "Brighten" linzasi orqali beriladi: +100 ochroq, -100 to'qroq.
Private Function ConvertShape(s As Shape, mode As cdrImageType, res As Long, bright As Long) As Boolean
    Dim target As Shape, r As Shape, grp As ShapeRange
    Dim aa As cdrAntiAliasingType
    On Error GoTo Fail

    Set target = s
    If bright <> 0 Then
        Set r = s.Layer.CreateRectangle2(s.LeftX, s.BottomY, s.SizeWidth, s.SizeHeight)
        r.Outline.SetNoOutline
        r.CreateLens cdrLensBrighten, bright
        r.OrderFrontOf s
        Set grp = CreateShapeRange
        grp.Add s
        grp.Add r
        Set target = grp.Group
    End If

    If chkAA.Value Then aa = cdrNormalAntiAliasing Else aa = cdrNoAntiAliasing
    target.ConvertToBitmapEx mode, False, CBool(chkTransp.Value), res, aa, True, False, 95
    ConvertShape = True
    Exit Function

Fail:
    ConvertShape = False
End Function

Private Sub SetStatus(msg As String)
    lblStatus.Caption = msg
    DoEvents
End Sub
