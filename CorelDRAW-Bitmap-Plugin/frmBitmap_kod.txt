' Segmentli tanlov: kulrang fon + oq "surgich" + yozuvlar
Private Sub BuildSeg(key As String, l As Single, t As Single, w As Single, caps As Variant, tips As Variant, _
                     lbls() As Object, thumb As MSForms.Image)
    Dim n As Long, i As Long, segW As Single, th As Single
    n = UBound(caps) + 1
    segW = (w - 4) / n
    th = CTL_H - 4
    AddImg l, t, w, CTL_H, RenderPic("seg", w, CTL_H, C_CARD, Array(0, 0, w, CTL_H, R_CTL, C_FIELD))
    Set thumb = AddImg(l + 2, t + 2, segW, th, RenderPic("thumb", segW, th, C_FIELD, _
                Array(0, 0, segW, th, R_CTL - 2, C_LINE), Array(Px1, Px1, segW - 2 * Px1, th - 2 * Px1, R_CTL - 2 - Px1, C_CARD)))
    For i = 0 To n - 1
        Set lbls(i) = AddLbl(CStr(caps(i)), l + 2 + i * segW, CenterTop(t, CTL_H, FS_BODY), segW, 16, FS_BODY, _
                             False, C_MUTED, fmTextAlignCenter, CStr(tips(i)))
    Next i
End Sub

Private Sub SelectSeg(lbls() As Object, thumb As MSForms.Image, idx As Long)
    Dim i As Long
    thumb.Left = lbls(idx).Left
    For i = LBound(lbls) To UBound(lbls)
        lbls(i).Font.Bold = (i = idx)
        lbls(i).ForeColor = IIf(i = idx, C_TEXT, C_MUTED)
    Next i
End Sub

' =====================================================================
'  YUMALOQ SHAKLLARNI CHIZISH (BMP, silliq chetlar bilan)
' =====================================================================
Private Function ScreenDpi() As Double
#If VBA7 Then
    Dim dc As LongPtr
#Else
    Dim dc As Long
#End If
    On Error GoTo Fallback
    dc = GetDC(0)
    ScreenDpi = GetDeviceCaps(dc, 88)   ' LOGPIXELSX
    ReleaseDC 0, dc
    If ScreenDpi < 72 Then ScreenDpi = 96
    Exit Function
Fallback:
    ScreenDpi = 96
End Function

' 1 piksel (pt da)
Private Function Px1() As Single
    Px1 = 1 / mK
End Function

' Har bir shakl: Array(x, y, w, h, radius, rang) - pt da.
Private Function RenderPic(key As String, w As Single, h As Single, bg As Long, ParamArray shapes() As Variant) As Object
    Dim path As String, wp As Long, hp As Long, rowSize As Long, n As Long
    Dim buf() As Byte, x As Long, y As Long, i As Long, off As Long, f As Integer
    Dim cx() As Double, cy() As Double, hx() As Double, hy() As Double, rr() As Double
    Dim sr() As Double, sg() As Double, sb() As Double
    Dim cr As Double, cg As Double, cb As Double, a As Double
    Dim qx As Double, qy As Double, ox As Double, oy As Double, m As Double, d As Double
    Dim s As Variant, r As Double, col As Long

    On Error GoTo Fail
    wp = Int(w * mK + 0.5): If wp < 1 Then wp = 1
    hp = Int(h * mK + 0.5): If hp < 1 Then hp = 1
    path = Environ$("TEMP") & "\bmpk_" & PIC_VER & "_" & key & "_" & wp & "x" & hp & ".bmp"

    If Len(Dir$(path)) = 0 Then
        n = UBound(shapes)
        If n >= 0 Then
            ReDim cx(n): ReDim cy(n): ReDim hx(n): ReDim hy(n): ReDim rr(n)
            ReDim sr(n): ReDim sg(n): ReDim sb(n)
        End If
        For i = 0 To n
            s = shapes(i)
            r = s(4) * mK
            If r > s(2) * mK / 2 Then r = s(2) * mK / 2
            If r > s(3) * mK / 2 Then r = s(3) * mK / 2
            If r < 0 Then r = 0
            cx(i) = (s(0) + s(2) / 2) * mK
            cy(i) = (s(1) + s(3) / 2) * mK
            hx(i) = s(2) * mK / 2 - r
            hy(i) = s(3) * mK / 2 - r
            rr(i) = r
            col = s(5)
            sr(i) = col And &HFF: sg(i) = (col \ &H100) And &HFF: sb(i) = (col \ &H10000) And &HFF
        Next i

        rowSize = ((wp * 3 + 3) \ 4) * 4
        ReDim buf(0 To 53 + rowSize * hp)
        buf(0) = 66: buf(1) = 77                 ' "BM"
        PutLong buf, 2, 54 + rowSize * hp
        PutLong buf, 10, 54
        PutLong buf, 14, 40
        PutLong buf, 18, wp
        PutLong buf, 22, hp
        buf(26) = 1: buf(28) = 24
        PutLong buf, 34, rowSize * hp
        PutLong buf, 38, 2835
        PutLong buf, 42, 2835

        For y = 0 To hp - 1
            For x = 0 To wp - 1
                cr = bg And &HFF: cg = (bg \ &H100) And &HFF: cb = (bg \ &H10000) And &HFF
                For i = 0 To n
                    qx = Abs(x + 0.5 - cx(i)) - hx(i)
                    qy = Abs(y + 0.5 - cy(i)) - hy(i)
                    If qx > 0 Then ox = qx Else ox = 0
                    If qy > 0 Then oy = qy Else oy = 0
                    If qx > qy Then m = qx Else m = qy
                    If m > 0 Then m = 0
                    d = Sqr(ox * ox + oy * oy) + m - rr(i)
                    a = 0.5 - d
                    If a > 0 Then
                        If a > 1 Then a = 1
                        cr = cr + (sr(i) - cr) * a
                        cg = cg + (sg(i) - cg) * a
                        cb = cb + (sb(i) - cb) * a
                    End If
                Next i
                off = 54 + (hp - 1 - y) * rowSize + x * 3
                buf(off) = CByte(cb): buf(off + 1) = CByte(cg): buf(off + 2) = CByte(cr)
            Next x
        Next y

        f = FreeFile
        Open path For Binary Access Write As #f
        Put #f, 1, buf
        Close #f
    End If

    Set RenderPic = LoadPicture(path)
    Exit Function
Fail:
    Set RenderPic = Nothing     ' rasm chizilmasa ham oyna ishlayveradi
End Function

Private Sub PutLong(buf() As Byte, pos As Long, ByVal v As Long)
    buf(pos) = v And &HFF
    buf(pos + 1) = (v \ &H100) And &HFF
    buf(pos + 2) = (v \ &H10000) And &HFF
    buf(pos + 3) = (v \ &H1000000) And &HFF
End Sub

' Ruxsat: tayyor qiymatlar bo'yicha qadam
Private Sub StepRes(dir As Long)
    Dim p As Variant, v As Long, i As Long
    p = Array(72, 150, 200, 300, 600, 1200, 2400)
    v = CLng(Val(txtRes.Text))
    If dir > 0 Then
        For i = 0 To UBound(p)
            If p(i) > v Then txtRes.Text = CStr(p(i)): Exit Sub
        Next i
    Else
        For i = UBound(p) To 0 Step -1
            If p(i) < v Then txtRes.Text = CStr(p(i)): Exit Sub
        Next i
    End If
End Sub

Private Sub StepBright(dir As Long)
    Dim v As Long
    v = CLng(Val(txtBright.Text)) + 5 * dir
    If v < -100 Then v = -100
    If v > 100 Then v = 100
    txtBright.Text = CStr(v)
End Sub

Private Sub ToggleSwitch(i As Long)
    mSw(i) = Not mSw(i)
    RefreshSwitch i
End Sub

Private Sub RefreshSwitch(i As Long)
    If picOn Is Nothing Or picOff Is Nothing Then
        ' rasm chizilmagan bo'lsa - oddiy rangli to'rtburchak
        swImg(i).BackStyle = fmBackStyleOpaque
        swImg(i).BackColor = IIf(mSw(i), C_ON, C_OFF)
    ElseIf mSw(i) Then
        Set swImg(i).Picture = picOn
    Else
        Set swImg(i).Picture = picOff
    End If
End Sub

' =====================================================================
Private Sub RunConvert()
    Dim res As Long, mode As cdrImageType, bright As Long
    Dim pg As Page, sr As ShapeRange
    Dim okCount As Long, errCount As Long

    If Documents.Count = 0 Then
        SetStatus "Hujjat ochilmagan", True
        Exit Sub
    End If

    res = CLng(Val(txtRes.Text))
    If Not IsNumeric(txtRes.Text) Or res < 30 Or res > 3000 Then
        SetStatus "Ruxsat: 30-3000 dpi", True
        Exit Sub
    End If

    If Not IsNumeric(txtBright.Text) Then
        SetStatus "Yorqinlik: -100 ... 100", True
        Exit Sub
    End If
    bright = CLng(Val(txtBright.Text))
    If bright < -100 Or bright > 100 Then
        SetStatus "Yorqinlik: -100 ... 100", True
        Exit Sub
    End If

    Select Case mColor
        Case 0: mode = cdrRGBColorImage
        Case 1: mode = cdrCMYKColorImage
        Case Else: mode = cdrGrayscaleImage
    End Select

    If mScope = 0 And ActiveSelectionRange.Count = 0 Then
        SetStatus "Hech narsa tanlanmagan", True
        Exit Sub
    End If

    mBusy = True
    lApply.Caption = "Ishlanmoqda" & ChrW(&H2026)
    ActiveDocument.BeginCommandGroup "Bitmap"
    Optimization = True
    On Error GoTo Finish

    If mScope = 0 Then
        Set sr = FilterRange(ActiveSelectionRange)
        ProcessRange sr, mode, res, bright, okCount, errCount
    ElseIf mScope = 1 Then
        Set sr = CollectPage(ActivePage)
        ProcessRange sr, mode, res, bright, okCount, errCount
    Else
        For Each pg In ActiveDocument.Pages
            pg.Activate
            SetStatus "Sahifa " & pg.Index & " / " & ActiveDocument.Pages.Count
            Set sr = CollectPage(pg)
            ProcessRange sr, mode, res, bright, okCount, errCount
        Next pg
    End If

Finish:
    If Err.Number <> 0 Then errCount = errCount + 1
    On Error GoTo 0
    Optimization = False
    ActiveDocument.EndCommandGroup
    ActiveWindow.Refresh
    Application.Refresh
    mBusy = False
    lApply.Caption = T("Qo`llash")

    If errCount > 0 Then
        SetStatus "Bajarildi: " & okCount & "  " & ChrW(&HB7) & "  Xato: " & errCount, True
        MsgBox "Xato: " & errCount & " ta obyekt o" & ChrW(&H2018) & "zgarmadi.", vbExclamation, Me.Caption
    Else
        SetStatus "Bajarildi: " & okCount
    End If
End Sub

Private Function CollectPage(pg As Page) As ShapeRange
    Dim sr As ShapeRange, lyr As Layer, s As Shape
    Set sr = CreateShapeRange
    For Each lyr In pg.Layers
        If Not lyr.IsSpecialLayer And lyr.Visible And lyr.Editable Then
            For Each s In lyr.Shapes
                If IsEligible(s) Then sr.Add s
            Next s
        End If
    Next lyr
    Set CollectPage = sr
End Function

Private Function FilterRange(src As ShapeRange) As ShapeRange
    Dim sr As ShapeRange, s As Shape
    Set sr = CreateShapeRange
    For Each s In src
        If IsEligible(s) Then sr.Add s
    Next s
    Set FilterRange = sr
End Function

Private Function IsEligible(s As Shape) As Boolean
    If s.Type = cdrGuidelineShape Then Exit Function
    If s.Locked Then Exit Function
    IsEligible = True
End Function

Private Sub ProcessRange(sr As ShapeRange, mode As cdrImageType, res As Long, bright As Long, _
                         okCount As Long, errCount As Long)
    Dim i As Long
    If sr.Count = 0 Then Exit Sub

    If (mMethod = 1) Then
        If sr.Count > 1 Then
            If ConvertShape(sr.Group, mode, res, bright) Is Nothing Then errCount = errCount + 1 Else okCount = okCount + 1
        Else
            If ConvertShape(sr(1), mode, res, bright) Is Nothing Then errCount = errCount + 1 Else okCount = okCount + 1
        End If
    Else
        For i = 1 To sr.Count
            SetStatus "Ishlanmoqda: " & i & " / " & sr.Count
            ProcessShape sr(i), mode, res, bright, okCount, errCount
        Next i
    End If
End Sub

' Obyektni ishlash:
'  - Guruh: "Guruhni butunligicha bitta rasm qilish" yoqilgan bo'lsa - butun guruh (ichidagi
'    rasm, vektor, PowerClip bilan) bitta bitmap bo'ladi; o'chirilgan bo'lsa - guruh ochiladi,
'    har bir bo'lagi alohida bitmap qilinadi va yana guruhlanadi.
'  - PowerClip ("PowerClip ichidagilarni ham" yoqilgan bo'lsa): ichidagilar chiqariladi,
'    ishlanadi va joyiga qaytariladi (konteyner ramkasi o'zgarmaydi).
' Natija sifatida hosil bo'lgan obyekt(lar) qaytariladi.
Private Function ProcessShape(s As Shape, mode As cdrImageType, res As Long, bright As Long, _
                              okCount As Long, errCount As Long) As ShapeRange
    Dim out As ShapeRange, kids As ShapeRange, inner As ShapeRange
    Dim pc As PowerClip, b As Shape, nm As String, i As Long

    Set out = CreateShapeRange
    Set ProcessShape = out
    On Error GoTo Fail

    If mSw(SW_DEEP) Then
        ' ----- PowerClip ichidagilar -----
        On Error Resume Next
        Set pc = s.PowerClip
        On Error GoTo Fail
        If Not pc Is Nothing Then
            If pc.Shapes.Count > 0 Then
                Set kids = pc.ExtractShapes
                Set inner = CreateShapeRange
                For i = 1 To kids.Count
                    inner.AddRange ProcessShape(kids(i), mode, res, bright, okCount, errCount)
                Next i
                If inner.Count > 0 Then inner.AddToPowerClip s, cdrFalse
            End If
            out.Add s
            Exit Function
        End If
    End If

    ' ----- Guruhni bo'laklab ishlash (bitta rasm qilish o'chirilgan bo'lsa) -----
    If Not mSw(SW_GROUP) Then
        If s.Type = cdrGroupShape Then
            nm = s.Name
            Set kids = s.UngroupEx
            Set inner = CreateShapeRange
            For i = 1 To kids.Count
                inner.AddRange ProcessShape(kids(i), mode, res, bright, okCount, errCount)
            Next i
            If inner.Count > 1 Then
                Set b = inner.Group
                If nm <> "" Then b.Name = nm
                out.Add b
            Else
                out.AddRange inner
            End If
            Exit Function
        End If
    End If

    ' ----- Oddiy obyekt (rasm, vektor, matn ...) -----
    ' (Guruh shu yerga kelsa - butunligicha bitta bitmap bo'ladi)
    If s.Locked Or s.Type = cdrGuidelineShape Or _
       (mSw(SW_ONLYBMP) And s.Type <> cdrBitmapShape And s.Type <> cdrGroupShape) Then
        out.Add s
        Exit Function
    End If

    Set b = ConvertShape(s, mode, res, bright)
    If b Is Nothing Then
        errCount = errCount + 1
        out.Add s
    Else
        okCount = okCount + 1
        out.Add b
    End If
    Exit Function

Fail:
    errCount = errCount + 1
    On Error Resume Next
    out.Add s
End Function

' Bitta obyektni bitmapga aylantirish.
' Yorqinlik "Brighten" linzasi orqali beriladi: +100 ochroq, -100 to'qroq.
Private Function ConvertShape(s As Shape, mode As cdrImageType, res As Long, bright As Long) As Shape
    Dim target As Shape, r As Shape, grp As ShapeRange
    Dim aa As cdrAntiAliasingType
    On Error GoTo Fail

    Set target = s
    If bright <> 0 Then
        Set r = s.Layer.CreateRectangle2(s.LeftX, s.BottomY, s.SizeWidth, s.SizeHeight)
        r.Outline.SetNoOutline
        r.CreateLens cdrLensBrighten, bright
        r.OrderFrontOf s
        Set grp = CreateShapeRange
        grp.Add s
        grp.Add r
        Set target = grp.Group
    End If

    If mSw(SW_AA) Then aa = cdrNormalAntiAliasing Else aa = cdrNoAntiAliasing
    Set ConvertShape = target.ConvertToBitmapEx(mode, False, mSw(SW_TRANSP), res, aa, True, False, 95)
    Exit Function

Fail:
    Set ConvertShape = Nothing
End Function
